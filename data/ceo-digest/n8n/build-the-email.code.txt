const nf = (o) => new Intl.NumberFormat(loc, o);
const dash = (v) => v === null || v === undefined || v === '';
const eur = (v) => dash(v) ? '–' : nf({ style: 'currency', currency: 'EUR', maximumFractionDigits: 0 }).format(Number(v));
const eur2 = (v) => dash(v) ? '–' : nf({ style: 'currency', currency: 'EUR', minimumFractionDigits: 2, maximumFractionDigits: 2 }).format(Number(v));
const num = (v) => dash(v) ? '–' : nf({ maximumFractionDigits: 1 }).format(Number(v));

const esc = (s) => String(s).replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));

function table(cols, rows) {
  if (!Array.isArray(rows) || !rows.length) return '<p style="color:#6b7280;margin:6px 0 18px">' + esc(L.none) + '</p>';
  const cell = (c) => 'padding:6px 8px;text-align:' + (c.r ? 'right' : 'left') + ';' + (c.b ? 'font-weight:700;' : '');
  let t = '<table role="presentation" style="border-collapse:collapse;width:100%;font-size:13px;margin:6px 0 22px">';
  t += '<tr>' + cols.map((c) => '<th style="' + cell(c) + 'border-bottom:2px solid #0f172a;color:#334155;font-weight:600;white-space:nowrap">' + esc(c.h) + '</th>').join('') + '</tr>';
  rows.forEach((row, i) => {
    const code = String(row.channel_code || row.category_code || row.plan_code || '');
    const total = /^(total|total_in|total_out|net)$/.test(code);
    const bg = total ? 'background:#e2e8f0;font-weight:700;' : (i % 2 ? 'background:#f8fafc;' : '');
    t += '<tr style="' + bg + '">' + cols.map((c) => '<td style="' + cell(c) + 'border-bottom:1px solid #e2e8f0">' + esc(c.f(row[c.k])) + '</td>').join('') + '</tr>';
  });
  return t + '</table>';
}

// A calendar period by its name ("March 2026", "Q3 2026", "2025", "Week 46 2026"), else null.
function periodName(from, to) {
  const t = (s) => Date.parse(s + 'T00:00:00Z');
  const ymd = (ms) => new Date(ms).toISOString().slice(0, 10);
  const f = new Date(t(from)), y = f.getUTCFullYear(), m = f.getUTCMonth();
  const end = (months) => ymd(Date.UTC(y, m + months, 0));
  if (from.endsWith('-01') && to === end(1)) return f.toLocaleDateString(loc, { month: 'long', year: 'numeric', timeZone: 'UTC' });
  if (from.endsWith('-01') && m % 3 === 0 && to === end(3)) return L.top.quarter(m / 3 + 1, y);
  if (from.endsWith('-01') && m === 0 && to === end(12)) return L.top.year(y);
  if (f.getUTCDay() === 1 && t(to) - t(from) === 6 * 86400000) {
    const th = new Date(t(from) + 3 * 86400000), wy = th.getUTCFullYear();
    return L.top.week(Math.floor((th - Date.UTC(wy, 0, 1)) / (7 * 86400000)) + 1, wy);
  }
  return null;
}
